   178→function getHighlights(product: Product): string[] {
   179→  const preset = HIGHLIGHTS[product.slug];
   180→  if (preset && preset.length > 0) return preset;
   181→  // Запасной вариант: первые предложения описания товара
   182→  return product.description
   183→    .split(". ")
   184→    .map((part) => part.trim())
   185→    .filter(Boolean)
   186→    .map((part) => (part.endsWith(".") ? part : `${part}.`))
   187→    .slice(0, 3);
   188→}

   204→export function ProductPage({
   205→  productId,
   206→  onAdd,
   207→  onQuickAdd,
   208→  onOpenProduct,
   209→  onGoCategory,
   210→  onGoHome,
   211→}: ProductPageProps) {
   212→  const [product, setProduct] = useState<Product | null>(null);
   213→  const [status, setStatus] = useState<"loading" | "ready" | "notfound">("loading");
   214→  const [related, setRelated] = useState<Product[]>([]);
   215→
   216→  useEffect(() => {
   217→    let cancelled = false;
   218→    // Компонент перемонтируется при смене товара (key={productId} в родителе),
   219→    // поэтому исходное состояние уже «loading» — здесь только загрузка данных.
   220→
   221→    fetch(`/api/products/${productId}`)
   222→      .then((res) => (res.ok ? res.json() : res.status === 404 ? null : Promise.reject(new Error("bad response"))))
   223→      .then((data: Product | null) => {
   224→        if (cancelled) return;
   225→        if (!data) {
   226→          setStatus("notfound");
   227→          return;
   228→        }
   229→        setProduct(data);
   230→        setStatus("ready");
   231→        window.scrollTo({ top: 0 });
   232→        // Похожие: сначала своя категория; если меньше 8 — добираем популярными
   233→        // из других категорий, чтобы карусель не выглядела полупустой
   234→        fetch(`/api/products?category=${data.category.slug}&sort=popular`)
   235→          .then((res) => (res.ok ? res.json() : Promise.reject(new Error("bad response"))))
   236→          .then(async (list: Product[]) => {
   237→            if (cancelled) return;
   238→            const sameCategory = list.filter((item) => item.id !== data.id).slice(0, 8);
   239→            setRelated(sameCategory);
   240→            if (sameCategory.length < 8) {
   241→              const res = await fetch("/api/products?sort=popular");
   242→              if (!res.ok || cancelled) return;
   243→              const popular = (await res.json()) as Product[];
   244→              const used = new Set([data.id, ...sameCategory.map((item) => item.id)]);
   245→              const extra = popular.filter((item) => !used.has(item.id)).slice(0, 8 - sameCategory.length);
   246→              setRelated([...sameCategory, ...extra]);
   247→            }
   248→          })
   249→          .catch(() => {});
   250→      })
   251→      .catch(() => {
   252→        if (!cancelled) setStatus("notfound");
   253→      });
   254→
   255→    return () => {
   256→      cancelled = true;
   257→    };
   258→  }, [productId]);
   259→
   260→  // Заголовок вкладки — имя товара (восстанавливаем при уходе)
   261→  useEffect(() => {
   262→    if (!product) return;
   263→    const previous = document.title;
   264→    document.title = `${product.name} — КуклаМаркет`;
   265→    return () => {
   266→      document.title = previous;
   267→    };
   268→  }, [product]);
   269→
   270→  if (status === "loading") {
   271→    return (
   272→      <div className="shop-container pb-6 pt-4 sm:pt-6" aria-busy="true" aria-label="Загружаем товар">
   273→        <Skeleton className="h-4 w-64 max-w-full" />
   274→        <div className="mt-6 grid items-start gap-8 lg:grid-cols-2 lg:gap-12">
   275→          <div>
   276→            <Skeleton className="aspect-square w-full rounded-xl" />
   277→            <div className="mt-3 flex gap-2.5">
   278→              {Array.from({ length: 3 }, (_, index) => (
   279→                <Skeleton key={index} className="size-16 rounded-lg sm:size-20" />
   280→              ))}
   281→            </div>
   282→          </div>
   283→          <div className="space-y-4">
   284→            <Skeleton className="h-4 w-44" />
   285→            <Skeleton className="h-9 w-11/12" />
   286→            <Skeleton className="h-5 w-32" />
   287→            <Skeleton className="h-10 w-52" />
   288→            <Skeleton className="h-12 w-full max-w-sm rounded-full" />
   289→          </div>
   290→        </div>
   291→      </div>
   292→    );
   293→  }
   294→
   295→  if (status === "notfound" || !product) {
   296→    return (
   297→      <div className="shop-container flex flex-col items-center gap-3 pb-12 pt-20 text-center">
   298→        <p className="text-2xl font-extrabold text-gray-900">Товар не найден</p>
   299→        <p className="max-w-sm text-sm text-gray-500">
   300→          Возможно, он уже распродан или ссылка устарела. Загляните в каталог — там много других кукол!
   301→        </p>
   302→        <button
   303→          type="button"
   304→          onClick={onGoHome}
   305→          className="mt-2 rounded-full bg-[#CC0000] px-6 py-2.5 text-sm font-bold text-white transition-colors hover:bg-[#A80000]"
   306→        >
   307→          Вернуться в каталог
   308→        </button>
   309→      </div>
   310→    );
   311→  }
   312→
   313→  return (
   314→    <>
   315→      <div className="shop-container pt-4 sm:pt-6">
   316→        <ProductBreadcrumbs product={product} onGoHome={onGoHome} onGoCategory={onGoCategory} />
   317→
   318→        <div className="mt-4 grid items-start gap-8 sm:mt-6 lg:grid-cols-2 lg:gap-12">
   319→          <ProductGallery product={product} />
   320→          <ProductInfo key={product.id} product={product} onAdd={onAdd} onGoCategory={onGoCategory} />
   321→        </div>
   322→      </div>
   323→
   324→      <div className="pb-6">
   325→        <AboutItemSection product={product} />
   326→
   327→        {related.length > 0 && (
   328→          <section className="shop-container mt-10 sm:mt-14" aria-label="Похожие товары">
   329→            <SectionHeader
   330→              title="Похожие товары"
   331→              subtitle={`Из категории «${product.category.name}» и не только — что ещё смотрят покупатели`}
   332→            />
   333→            <Carousel opts={{ align: "start", slidesToScroll: 2 }} className="mt-5">
   334→              <CarouselContent className="-ml-3 sm:-ml-4">
   335→                {related.map((item) => (
   336→                  <CarouselItem
   337→                    key={item.id}
   338→                    className="basis-1/2 pl-3 sm:basis-1/3 sm:pl-4 lg:basis-1/4 xl:basis-1/5"
   339→                  >
   340→                    <ProductCard product={item} onOpen={onOpenProduct} onAdd={onQuickAdd} />
   341→                  </CarouselItem>
   342→                ))}
   343→              </CarouselContent>
   344→              <CarouselPrevious
   345→                aria-label="Предыдущие товары"
   346→                className="hidden shadow-md sm:flex sm:-left-6 lg:-left-8"
   347→              />
   348→              <CarouselNext
   349→                aria-label="Следующие товары"
   350→                className="hidden shadow-md sm:flex sm:-right-6 lg:-right-8"
   351→              />
   352→            </Carousel>
   353→          </section>
   354→        )}
   355→      </div>
   356→    </>
   357→  );
   358→}
