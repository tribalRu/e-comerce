    43→export interface CategoryFilters {

    66→export function isCategoryFiltersActive(filters: CategoryFilters): boolean {
    67→  return (
    68→    filters.brands.length > 0 ||
    69→    filters.ages.length > 0 ||
    70→    filters.offers.length > 0 ||
    71→    filters.minRating > 0 ||
    72→    filters.priceMin != null ||
    73→    filters.priceMax != null
    74→  );
    75→}

    77→/** Отфильтровать товары категории по текущим значениям (внутри группы — ИЛИ) */
    78→export function applyCategoryFilters(
    79→  products: Product[],
    80→  filters: CategoryFilters,
    81→): Product[] {
    82→  return products.filter((product) => {
    83→    if (filters.brands.length > 0 && !filters.brands.includes(product.brand)) return false;
    84→    if (filters.ages.length > 0 && !filters.ages.includes(product.ageMin)) return false;
    85→    if (filters.offers.length > 0 && !matchesOffers(product, filters.offers)) return false;
    86→    if (filters.minRating > 0 && product.rating < filters.minRating) return false;
    87→    if (filters.priceMin != null && product.price < filters.priceMin) return false;
    88→    if (filters.priceMax != null && product.price > filters.priceMax) return false;
    89→    return true;
    90→  });
    91→}
    92→
    93→function matchesOffers(product: Product, offers: string[]): boolean {
    94→  return offers.some((offer) =>
    95→    offer === "discounted" ? product.oldPrice != null : product.badge === offer,
    96→  );
    97→}

    99→/** Клиентская реплика SORT_ORDERS из GET /api/products — смена сортировки
   100→ *  не перезагружает список, фильтры и порядок применяются мгновенно */
   101→export function sortCategoryProducts(products: Product[], sort: string): Product[] {
   102→  const copy = [...products];
   103→  copy.sort((a, b) => {
   104→    switch (sort) {
   105→      case "price_asc":
   106→        return a.price - b.price;
   107→      case "price_desc":
   108→        return b.price - a.price;
   109→      case "rating":
   110→        return b.rating - a.rating || b.reviewsCount - a.reviewsCount;
   111→      case "new":
   112→        return createdAtMs(b) - createdAtMs(a) || b.id - a.id;
   113→      default:
   114→        // popular — как в API: по числу отзывов, затем по рейтингу
   115→        return b.reviewsCount - a.reviewsCount || b.rating - a.rating;
   116→    }
   117→  });
   118→  return copy;
   119→}
   120→
   121→function createdAtMs(product: Product): number {
   122→  const time = Date.parse(product.createdAt ?? "");
   123→  return Number.isFinite(time) ? time : 0;
   124→}

   127→// Фасеты: доступные значения и счётчики по всем товарам категории
   128→
   129→const OFFER_KEYS = ["discounted", "hit", "new", "deal"] as const;
   130→
   131→const OFFER_LABELS: Record<string, string> = {
   132→  discounted: "Со скидкой",
   133→  hit: "Хит продаж",
   134→  new: "Новинка",
   135→  deal: "Выгодная цена",
   136→};

   138→const RATING_OPTIONS = [4.5, 4, 3];
   139→
   140→interface Facets {
   141→  total: number;
   142→  brands: { value: string; label: string; count: number }[];
   143→  ages: { value: number; label: string; count: number }[];
   144→  offers: { value: string; label: string; count: number }[];
   145→  ratings: { value: number; count: number }[];
   146→  /** Округлённые до сотен границы цен категории */
   147→  priceMin: number;
   148→  priceMax: number;
   149→}
   150→
   151→function buildFacets(products: Product[]): Facets {
   152→  const brands = new Map<string, number>();
   153→  const ages = new Map<number, number>();
   154→  const offers = new Map<string, number>();
   155→  for (const product of products) {
   156→    brands.set(product.brand, (brands.get(product.brand) ?? 0) + 1);
   157→    ages.set(product.ageMin, (ages.get(product.ageMin) ?? 0) + 1);
   158→    if (product.oldPrice != null) {
   159→      offers.set("discounted", (offers.get("discounted") ?? 0) + 1);
   160→    }
   161→    if (product.badge) {
   162→      offers.set(product.badge, (offers.get(product.badge) ?? 0) + 1);
   163→    }
   164→  }
   165→  const prices = products.map((product) => product.price);
   166→  return {
   167→    total: products.length,
   168→    brands: [...brands.entries()]
   169→      .sort((a, b) => b[1] - a[1] || a[0].localeCompare(b[0], "ru"))
   170→      .map(([value, count]) => ({ value, label: value, count })),
   171→    ages: [...ages.entries()]
   172→      .sort((a, b) => a[0] - b[0])
   173→      .map(([value, count]) => ({ value, label: ageLabel(value), count })),
   174→    offers: OFFER_KEYS.filter((key) => offers.has(key)).map((key) => ({
   175→      value: key,
   176→      label: OFFER_LABELS[key],
   177→      count: offers.get(key) ?? 0,
   178→    })),
   179→    ratings: RATING_OPTIONS.map((value) => ({
   180→      value,
   181→      count: products.filter((product) => product.rating >= value).length,
   182→    })),
   183→    priceMin: prices.length > 0 ? Math.floor(Math.min(...prices) / 100) * 100 : 0,
   184→    priceMax: prices.length > 0 ? Math.ceil(Math.max(...prices) / 100) * 100 : 0,
   185→  };
   186→}
   187→
   188→function ageLabel(age: number): string {
   189→  return `${age}+ ${pluralize(age, "год", "года", "лет")}`;
   190→}
   191→
   192→function formatRating(value: number): string {
   193→  return value.toLocaleString("ru-RU");
   194→}

   318→          />
   319→        ))}

   323→

   639→/** Диапазон цены: ползунок и поля «от/до» с живой фильтрацией */
   640→function PriceRangeControl({
   641→  facets,
   642→  filters,
   643→  onFiltersChange,
   644→}: {
   645→  facets: Facets;
   646→  filters: CategoryFilters;
   647→  onFiltersChange: (filters: CategoryFilters) => void;
   648→}) {
   649→  const from = filters.priceMin ?? facets.priceMin;
   650→  const to = filters.priceMax ?? facets.priceMax;
   651→
   652→  // Черновики полей ввода: null — показывать значение из фильтра.
   653→  // Коммит по blur/Enter, чтобы набор числа не дёргал фильтр на каждой цифре
   654→  const [fromDraft, setFromDraft] = useState<string | null>(null);
   655→  const [toDraft, setToDraft] = useState<string | null>(null);
   656→  const shownFrom = fromDraft ?? (filters.priceMin != null ? filters.priceMin.toLocaleString("ru-RU") : "");
   657→  const shownTo = toDraft ?? (filters.priceMax != null ? filters.priceMax.toLocaleString("ru-RU") : "");
   658→
   659→  const commit = () => {
   660→    const rawFrom = fromDraft != null ? fromDraft : String(filters.priceMin ?? "");
   661→    const rawTo = toDraft != null ? toDraft : String(filters.priceMax ?? "");
   662→    let nextFrom = parsePriceText(rawFrom, facets);
   663→    let nextTo = parsePriceText(rawTo, facets);
   664→    if (nextFrom != null && nextTo != null && nextFrom > nextTo) {
   665→      [nextFrom, nextTo] = [nextTo, nextFrom];
   666→    }
   667→    onFiltersChange({
   668→      ...filters,
   669→      priceMin: nextFrom === facets.priceMin ? null : nextFrom,
   670→      priceMax: nextTo === facets.priceMax ? null : nextTo,
   671→    });
   672→    setFromDraft(null);
   673→    setToDraft(null);
   674→  };
   675→
   676→  return (
   677→    <div className="px-2">
   678→      <div className="flex items-center gap-2.5">
   679→        <Input
   680→          aria-label="Цена от"
   681→          inputMode="numeric"
   682→          placeholder={facets.priceMin.toLocaleString("ru-RU")}
   683→          value={shownFrom}
   684→          onChange={(event) => setFromDraft(event.target.value)}
   685→          onBlur={commit}
   686→          onKeyDown={(event) => {
   687→            if (event.key === "Enter") event.currentTarget.blur();
   688→          }}
   689→          className="h-9 text-sm"
   690→        />
   691→        <span className="shrink-0 text-sm text-gray-400" aria-hidden="true">
   692→          —
   693→        </span>
   694→        <Input
   695→          aria-label="Цена до"
   696→          inputMode="numeric"
   697→          placeholder={facets.priceMax.toLocaleString("ru-RU")}
   698→          value={shownTo}
   699→          onChange={(event) => setToDraft(event.target.value)}
   700→          onBlur={commit}
   701→          onKeyDown={(event) => {
   702→            if (event.key === "Enter") event.currentTarget.blur();
   703→          }}
   704→          className="h-9 text-sm"
   705→        />
   706→      </div>
   707→      <Slider
   708→        aria-label="Диапазон цены"
   709→        className="mt-4"
   710→        min={facets.priceMin}
   711→        max={facets.priceMax}
   712→        step={100}
   713→        value={[from, to]}
   714→        onValueChange={(values) => {
   715→          const [nextFrom, nextTo] = values;
   716→          onFiltersChange({
   717→            ...filters,
   718→            priceMin: nextFrom === facets.priceMin ? null : nextFrom,
   719→            priceMax: nextTo === facets.priceMax ? null : nextTo,
   720→          });
   721→        }}
   722→      />
   723→      <p className="mt-2.5 text-xs text-gray-400" aria-live="polite">
   724→        В этой категории: от {formatPrice(facets.priceMin)} до {formatPrice(facets.priceMax)}
   725→      </p>
   726→    </div>
   727→  );
   728→}
   729→
   730→/** Разобрать текст поля цены: только цифры, с ограничением по границам */
   731→function parsePriceText(text: string, facets: Facets): number | null {
   732→  const digits = text.replace(/\D/g, "");
   733→  if (!digits) return null;
   734→  return clamp(Number.parseInt(digits, 10), facets.priceMin, facets.priceMax);
   735→}

   737→function toggleInList<T>(list: T[], value: T): T[] {
   738→  return list.includes(value) ? list.filter((item) => item !== value) : [...list, value];
   739→}
