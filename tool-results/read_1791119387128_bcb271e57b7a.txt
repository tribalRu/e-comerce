   105→/** Целое из строки с зажимом в диапазон (некорректное → fallback) */
   106→function toInt(value: string, fallback: number, min: number, max: number): number {
   107→  const parsed = Number.parseInt(value, 10);
   108→  if (!Number.isFinite(parsed)) return fallback;
   109→  return Math.min(max, Math.max(min, parsed));
   110→}
   111→
   112→/** Дробное из строки с зажимом в диапазон и округлением до 1 знака */
   113→function toNum(value: string, fallback: number, min: number, max: number): number {
   114→  const parsed = Number.parseFloat(value.replace(",", "."));
   115→  if (!Number.isFinite(parsed)) return fallback;
   116→  const clamped = Math.min(max, Math.max(min, parsed));
   117→  return Math.round(clamped * 10) / 10;
   118→}
