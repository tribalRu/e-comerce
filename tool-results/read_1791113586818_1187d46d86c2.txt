   183→function getHighlights(product: Product): string[] {
   184→  const preset = HIGHLIGHTS[product.slug];
   185→  if (preset && preset.length > 0) return preset;
   186→  // Запасной вариант: первые предложения описания товара
   187→  return product.description
   188→    .split(". ")
   189→    .map((part) => part.trim())
   190→    .filter(Boolean)
   191→    .map((part) => (part.endsWith(".") ? part : `${part}.`))
   192→    .slice(0, 3);
   193→}

   195→// Заголовок аккордеона как на референсе: жирный текст слева, шеврон справа,

   200→interface ProductPageProps {
   201→  productId: number;
   202→  /** false — корзина выключена в админке: скрываем степпер, кнопку и липкую панель */
   203→  cartEnabled: boolean;
   204→  onAdd: (product: Product, quantity: number) => void;
   205→  onQuickAdd: (product: Product) => void;
   206→  onOpenProduct: (product: Product) => void;
   207→  onGoCategory: (slug: string) => void;
   208→  onGoHome: () => void;
   209→}
   210→
   211→export function ProductPage({
   212→  productId,
   213→  cartEnabled,
   214→  onAdd,
   215→  onQuickAdd,
   216→  onOpenProduct,
   217→  onGoCategory,
   218→  onGoHome,
   219→}: ProductPageProps) {
   220→  const [product, setProduct] = useState<Product | null>(null);
   221→  const [status, setStatus] = useState<"loading" | "ready" | "notfound">("loading");
   222→  const [related, setRelated] = useState<Product[]>([]);
   223→
   224→  // Количество — общее для блока покупки и липкой панели (синхронный степпер)
   225→  const [quantity, setQuantity] = useState(1);
   226→  // Блок «цена + количество + Добавить в корзину», за которым следит скролл
   227→  const buyBoxRef = useRef<HTMLDivElement>(null);
   228→  const [barVisible, setBarVisible] = useState(false);
   229→
   230→  useEffect(() => {
   231→    let cancelled = false;
   232→    // Компонент перемонтируется при смене товара (key={productId} в родителе),
   233→    // поэтому исходное состояние уже «loading» — здесь только загрузка данных.
   234→
   235→    fetch(`/api/products/${productId}`)
   236→      .then((res) => (res.ok ? res.json() : res.status === 404 ? null : Promise.reject(new Error("bad response"))))
   237→      .then((data: Product | null) => {
   238→        if (cancelled) return;
   239→        if (!data) {
   240→          setStatus("notfound");
   241→          return;
   242→        }
   243→        setProduct(data);
   244→        setStatus("ready");
   245→        window.scrollTo({ top: 0 });
   246→        // Похожие: сначала своя категория; если меньше 8 — добираем популярными
   247→        // из других категорий, чтобы карусель не выглядела полупустой
   248→        fetch(`/api/products?category=${data.category.slug}&sort=popular`)
   249→          .then((res) => (res.ok ? res.json() : Promise.reject(new Error("bad response"))))
   250→          .then(async (list: Product[]) => {
   251→            if (cancelled) return;
   252→            const sameCategory = list.filter((item) => item.id !== data.id).slice(0, 8);
   253→            setRelated(sameCategory);
   254→            if (sameCategory.length < 8) {
   255→              const res = await fetch("/api/products?sort=popular");
   256→              if (!res.ok || cancelled) return;
   257→              const popular = (await res.json()) as Product[];
   258→              const used = new Set([data.id, ...sameCategory.map((item) => item.id)]);
   259→              const extra = popular.filter((item) => !used.has(item.id)).slice(0, 8 - sameCategory.length);
   260→              setRelated([...sameCategory, ...extra]);
   261→            }
   262→          })
   263→          .catch(() => {});
   264→      })
   265→      .catch(() => {
   266→        if (!cancelled) setStatus("notfound");
   267→      });
   268→
   269→    return () => {
   270→      cancelled = true;
   271→    };
   272→  }, [productId]);
   273→
   274→  // Заголовок вкладки — имя товара (восстанавливаем при уходе)
   275→  useEffect(() => {
   276→    if (!product) return;
   277→    const previous = document.title;
   278→    document.title = `${product.name} — КуклаМаркет`;
   279→    return () => {
   280→      document.title = previous;
   281→    };
   282→  }, [product]);
   283→
   284→  // Липкая панель покупки: показываем, когда блок с ценой и количеством
   285→  // полностью ушёл вверх за шапку (его низ выше низа шапки); скрываем,
   286→  // когда блок снова появляется из-под шапки. Высоту шапки берём из DOM,
   287→  // чтобы не зависеть от жёсткой константы.
   288→  useEffect(() => {
   289→    if (status !== "ready") return;
   290→    let raf = 0;
   291→    const update = () => {
   292→      raf = 0;
   293→      const el = buyBoxRef.current;
   294→      if (!el) return;
   295→      const header = document.querySelector("header");
   296→      const headerBottom = header ? header.getBoundingClientRect().bottom : 76;
   297→      setBarVisible(el.getBoundingClientRect().bottom < headerBottom);
   298→    };
   299→    const onScrollOrResize = () => {
   300→      if (!raf) raf = requestAnimationFrame(update);
   301→    };
   302→    update();
   303→    window.addEventListener("scroll", onScrollOrResize, { passive: true });
   304→    window.addEventListener("resize", onScrollOrResize);
   305→    return () => {
   306→      if (raf) cancelAnimationFrame(raf);
   307→      window.removeEventListener("scroll", onScrollOrResize);
   308→      window.removeEventListener("resize", onScrollOrResize);
   309→    };
   310→  }, [status]);
   311→
   312→  if (status === "loading") {
   313→    return (
   314→      <div className="shop-container pb-6 pt-4 sm:pt-6" aria-busy="true" aria-label="Загружаем товар">
   315→        <Skeleton className="h-4 w-64 max-w-full" />
   316→        <div className="mt-6 grid items-start gap-8 lg:grid-cols-2 lg:gap-12">
   317→          <div>
   318→            <Skeleton className="aspect-square w-full rounded-xl" />
   319→            <div className="mt-3 flex gap-2.5">
   320→              {Array.from({ length: 3 }, (_, index) => (
   321→                <Skeleton key={index} className="size-16 rounded-lg sm:size-20" />
   322→              ))}
   323→            </div>
   324→          </div>
   325→          <div className="space-y-4">
   326→            <Skeleton className="h-4 w-44" />
   327→            <Skeleton className="h-9 w-11/12" />
   328→            <Skeleton className="h-5 w-32" />
   329→            <Skeleton className="h-10 w-52" />
   330→            <Skeleton className="h-12 w-full max-w-sm rounded-full" />
   331→          </div>
   332→        </div>
   333→      </div>
   334→    );
   335→  }
   336→
   337→  if (status === "notfound" || !product) {
   338→    return (
   339→      <div className="shop-container flex flex-col items-center gap-3 pb-12 pt-20 text-center">
   340→        <p className="text-2xl font-extrabold text-gray-900">Товар не найден</p>
   341→        <p className="max-w-sm text-sm text-gray-500">
   342→          Возможно, он уже распродан или ссылка устарела. Загляните в каталог — там много других кукол!
   343→        </p>
   344→        <button
   345→          type="button"
   346→          onClick={onGoHome}
   347→          className="mt-2 rounded-full bg-[#CC0000] px-6 py-2.5 text-sm font-bold text-white transition-colors hover:bg-[#A80000]"
   348→        >
   349→          Вернуться в каталог
   350→        </button>
   351→      </div>
   352→    );
   353→  }
   354→
   355→  const stockMax = product ? Math.max(product.stock, 1) : 1;
   356→  const changeQuantity = (value: number) => setQuantity(clamp(value, 1, stockMax));
   357→
   358→  return (
   359→    <>
   360→      {/* Липкая панель покупки — появляется при прокрутке за блок с ценой.
   361→          Скрывается целиком, когда корзина выключена в админ-панели. */}
   362→      {cartEnabled && (
   363→        <StickyBuyBar
   364→          product={product}
   365→          visible={barVisible}
   366→          quantity={quantity}
   367→          onQuantityChange={changeQuantity}
   368→          onGoCategory={onGoCategory}
   369→        />
   370→      )}
   371→
   372→      <div className="shop-container pt-4 sm:pt-6">
   373→        <ProductBreadcrumbs product={product} onGoHome={onGoHome} onGoCategory={onGoCategory} />
   374→
   375→        <div className="mt-4 grid items-start gap-8 sm:mt-6 lg:grid-cols-2 lg:gap-12">
   376→          <ProductGallery product={product} />
   377→          <ProductInfo
   378→            product={product}
   379→            cartEnabled={cartEnabled}
   380→            quantity={quantity}
   381→            onQuantityChange={changeQuantity}
   382→            buyBoxRef={buyBoxRef}
   383→            onAdd={onAdd}
   384→            onGoCategory={onGoCategory}
   385→          />
   386→        </div>
   387→      </div>
   388→
   389→      <div className="pb-6">
   390→        <AboutItemSection product={product} />
   391→
   392→        {related.length > 0 && (
   393→          <section className="shop-container mt-10 sm:mt-14" aria-label="Похожие товары">
   394→            <SectionHeader
   395→              title="Похожие товары"
   396→              subtitle={`Из категории «${product.category.name}» и не только — что ещё смотрят покупатели`}
   397→            />
   398→            <Carousel opts={{ align: "start", slidesToScroll: 2 }} className="mt-5">
   399→              <CarouselContent className="-ml-3 sm:-ml-4">
   400→                {related.map((item) => (
   401→                  <CarouselItem
   402→                    key={item.id}
   403→                    className="basis-1/2 pl-3 sm:basis-1/3 sm:pl-4 lg:basis-1/4 xl:basis-1/5"
   404→                  >
   405→                    <ProductCard
   406→                      product={item}
   407→                      onOpen={onOpenProduct}
   408→                      onAdd={onQuickAdd}
   409→                      canAdd={cartEnabled}
   410→                    />
   411→                  </CarouselItem>
